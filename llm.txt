1. The File Buffer "Leftover"
After reading the width and height with fscanf, you are absolutely correct: there is a newline character (\n) still sitting in the buffer immediately after the height integer.

The Problem: fscanf(file, "%d %d", ...) stops reading exactly after it consumes the last digit of the height. The \n that follows that number remains in the file stream.

The Consequence: If your very first call to fgetc inside the loop doesn't account for that, it will read that \n as the very first character of your grid (the top-left pixel).

The Fix: You need to consume that leftover newline before starting your nested loops. A simple way is to add a fgetc(file); immediately after your fscanf call to "flush" that character out.

void create_minimap(Map *map, const char *output_filename) {
    FILE *file = fopen(output_filename, "wb"); // Important: use "wb" for binary mode
    if (!file) return;

    // PPM P6 Header
    fprintf(file, "P6\n%d %d\n255\n", map->width, map->height);

    for (int y = 0; y < map->height; y++) {
        for (int x = 0; x < map->width; x++) {
            char cell = map->grid[y][x];
            unsigned char r, g, b;

            // Define colors as bytes (0-255)
            if (cell == 'S') {
                r = 0;   g = 255; b = 0;   // Green
            } else if (cell == ' ') {
                r = 255; g = 255; b = 255; // White
            } else {
                r = 0;   g = 0;   b = 0;   // Black (Wall)
            }

            // Write the 3 bytes directly
            fputc(r, file);
            fputc(g, file);
            fputc(b, file);
        }
    }
    fclose(file);

/* Main rendering */
void render_maze(Maze *maze,
                 const char *output_file,
                 int width,
                 int height,
                 double fov,
                 double rotation,
                 double px,
                 double py)
{
    /* Create output image */
    Image *img = create_image(width, height);

    /* Convert angles to radians */
    double alpha = rotation * PI / 180.0;
    double fov_rad = fov * PI / 180.0;

    /* Direction vector */
    double dx = cos(alpha);
    double dy = sin(alpha);

    /* Camera height */
    double pz = 0.5;

    /* Focal distance */
    double f = width / (2.0 * tan(fov_rad / 2.0));

    /* Principle point */
    double cx = px - f * dx;
    double cy = py - f * dy;

    /* Render every column */
    for (int x = 0; x < width; x++)
    {
        /*
         * Image plane coordinate
         * Formula from slides
         */
        double ix =
            cx + (x - width / 2.0 + 0.5) * dy;

        double iy =
            cy - (x - width / 2.0 + 0.5) * dx;

        /* Ray direction */
        double rx = px - ix;
        double ry = py - iy;

        /* Normalize direction */
        double length = sqrt(rx * rx + ry * ry);

        double ux = rx / length;
        double uy = ry / length;

        /* DDA variables */
        int mapX = (int)px;
        int mapY = (int)py;

        double sideDistX;
        double sideDistY;

        double deltaDistX = fabs(1.0 / ux);
        double deltaDistY = fabs(1.0 / uy);

        int stepX;
        int stepY;

        int hit = 0;
        int side;

        /* Step direction X */
        if (ux < 0)
        {
            stepX = -1;
            sideDistX =
                (px - mapX) * deltaDistX;
        }
        else
        {
            stepX = 1;
            sideDistX =
                (mapX + 1.0 - px) * deltaDistX;
        }

        /* Step direction Y */
        if (uy < 0)
        {
            stepY = -1;
            sideDistY =
                (py - mapY) * deltaDistY;
        }
        else
        {
            stepY = 1;
            sideDistY =
                (mapY + 1.0 - py) * deltaDistY;
        }

        /* DDA algorithm */
        while (!hit)
        {
            /* Move in smallest direction */
            if (sideDistX < sideDistY)
            {
                sideDistX += deltaDistX;
                mapX += stepX;
                side = 0;
            }
            else
            {
                sideDistY += deltaDistY;
                mapY += stepY;
                side = 1;
            }

            /* Check wall */
            if (is_wall(maze, mapX, mapY))
            {
                hit = 1;
            }
        }

        /* Calculate distance */
        double perpWallDist;

        if (side == 0)
        {
            perpWallDist =
                sideDistX - deltaDistX;
        }
        else
        {
            perpWallDist =
                sideDistY - deltaDistY;
        }

        /*
         * Wall height calculation
         */
        int lineHeight =
            (int)(height / perpWallDist);

        int drawStart =
            -lineHeight / 2 + height / 2;

        int drawEnd =
            lineHeight / 2 + height / 2;

        /* Clamp */
        if (drawStart < 0)
        {
            drawStart = 0;
        }

        if (drawEnd >= height)
        {
            drawEnd = height - 1;
        }

        /* Draw wall */
        for (int y = drawStart;
             y <= drawEnd;
             y++)
        {
            /* x-side wall = green */
            if (side == 0)
            {
                set_pixel(img,
                          x,
                          y,
                          0,
                          255,
                          0);
            }
            else
            {
                /* y-side wall = red */
                set_pixel(img,
                          x,
                          y,
                          255,
                          0,
                          0);
            }
        }
    }

    /* Save image */
    write_ppm(img, output_file);

    /* Cleanup */
    free_image(img);
}

In your original is_wall function, you returned 1 if a ray went outside the map boundaries. This forces the ray to register a "hit" on an invisible wall at the edge of the map.
